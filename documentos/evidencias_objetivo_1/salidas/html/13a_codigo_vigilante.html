1"""Vigilante continuo del JSON (python -m app.etl.watch_red_rooster)."""
2import logging
3import time
4
5from app.core.config import get_settings, obtener_clave_privacidad
6from app.core.db import SessionLocal
7from app.etl.red_rooster import importar
8
9logging.basicConfig(level=logging.INFO, format="%(asctime)s %(levelname)s %(message)s")

11
12
13def main() -> None:
14 cfg = get_settings()
15 intervalo = max(5, cfg.red_rooster_poll_seconds)

17 while True:
18 try:
19 with SessionLocal() as session:

22 log.exception("Falló la carga; se reintentará en %s segundos", intervalo)
23 time.sleep(intervalo)
24
25
26if __name__ == "__main__":
27 try:
28 main()

30 log.info("Vigilante detenido")
