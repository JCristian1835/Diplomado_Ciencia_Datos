37def _huella_cliente(pedido: dict[str, Any], clave: str) -> str | None:
38 """HMAC del teléfono (solo dígitos, sin el 57 del país) o, si no hay, de la dirección de un domicilio."""
39 identificador = re.sub(r"\D", "", str(pedido.get("telefono") or ""))
40 if len(identificador) == 12 and identificador.startswith("57"):
41 identificador = identificador[2:]
42 if not identificador and canal_de(pedido) == "domicilio":
43 identificador = " ".join(str(pedido.get("direccion") or "").lower().split())
44 if not identificador:
45 return None
46 return hmac.new(clave.encode(), identificador.encode(), hashlib.sha256).hexdigest()
