37def puntaje(serie, invertir=False):
38 """Puntaje 1-5 por quintiles: 1 + número de cortes (20, 40, 60, 80 %) que el valor supera.
39
40 Con empates (muchos clientes con 1 pedido) los quintiles se agrupan en menos niveles.
41 Si `invertir`, el menor valor recibe el puntaje mayor (recencia).
42 """
43 cortes = serie.quantile([0.2, 0.4, 0.6, 0.8]).to_numpy()
44 p = 1 + (serie.to_numpy()[:, None] > cortes).sum(axis=1)
45 return pd.Series(6 - p if invertir else p, index=serie.index)

48def asignar_segmentos(rfm):
