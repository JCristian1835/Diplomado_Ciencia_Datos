74def _cifras_inexistentes(dato: str, existentes: set[float]) -> list[str]:
75 """Cifras citadas que no aparecen en el contexto (tolera redondeos y proporciones expresadas en %)."""
76 faltan = []
77 for m in _NUMERO.finditer(dato):
78 escala = {"mil": 1e3}.get((m.group(2) or "").lower(), 1e6 if m.group(2) else 1.0)
79 for valor, dec in _valores_numero(m.group(1)):
80 citado, tol = valor * escala, 0.5 * 10 ** -dec * escala
81 if any(abs(citado - x) <= tol or (x <= 1 and abs(citado - x * 100) <= tol) for x in existentes):
82 break
83 else:
84 faltan.append(m.group(1))
85 return faltan
