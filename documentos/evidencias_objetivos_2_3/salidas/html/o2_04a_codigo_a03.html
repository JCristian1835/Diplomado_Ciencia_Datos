35def mann_whitney(a, b):
36 """U de Mann-Whitney bilateral y delta de Cliff (positivo si `a` tiende a ser mayor que `b`)."""
37 r = stats.mannwhitneyu(a, b, alternative="two-sided")
38 delta = 2 * r.statistic / (len(a) * len(b)) - 1
39 return {"U": r.statistic, "p": r.pvalue, "delta_cliff": delta, "magnitud": magnitud(delta)}
