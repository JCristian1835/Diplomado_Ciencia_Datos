18def resumen_periodo(g, anomalo):
19 """Medias por día abierto de un periodo, separando domicilios, días normales y días anómalos."""
20 otros = g["pedidos"] - g["domicilios"]
21 normales = otros[~anomalo]
22 anomalos = otros[anomalo]
23 perdida = len(anomalos) / len(g) * (normales.mean() - anomalos.mean()) if len(anomalos) else 0.0
24 return {"dias": len(g), "dias_anomalos": int(anomalo.sum()), "pedidos_por_dia": g["pedidos"].mean(),
25 "domicilios_por_dia": g["domicilios"].mean(), "nivel_dias_normales": normales.mean(),
26 "perdida_dias_anomalos": perdida}

29def descomponer(c, mascara_anomala):
30 """Devuelve la descomposición (a, b, c) de la caída para una definición de día anómalo."""
31 base = resumen_periodo(c[c["periodo"] == "base"], mascara_anomala[c["periodo"] == "base"])
32 caida = resumen_periodo(c[c["periodo"] == "caida"], mascara_anomala[c["periodo"] == "caida"])
33 total = base["pedidos_por_dia"] - caida["pedidos_por_dia"]
34 partes = {"domicilios": base["domicilios_por_dia"] - caida["domicilios_por_dia"],
35 "dias_casi_sin_pedidos": caida["perdida_dias_anomalos"] - base["perdida_dias_anomalos"],
36 "demanda_dias_normales": base["nivel_dias_normales"] - caida["nivel_dias_normales"]}
37 return {"base": base, "caida": caida, "caida_total": total, "partes": partes,
38 "porcentajes": {k: v / total for k, v in partes.items()},
39 "cuadre": abs(sum(partes.values()) - total) < 1e-9}
