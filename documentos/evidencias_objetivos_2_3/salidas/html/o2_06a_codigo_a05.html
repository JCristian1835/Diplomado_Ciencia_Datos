67def pares(h, origenes, saltos, hasta):
68 """Filas (origen, objetivo) con sus variables. `hasta` limita el objetivo a lo ya conocido."""
69 o = np.repeat(origenes, len(saltos))
70 g = np.tile(saltos, len(origenes))
71 t = o + g
72 ok = (t <= hasta) & (t < len(h.fechas)) & h.abierto[np.minimum(t, len(h.fechas) - 1)] & ~np.isnan(h.nivel[o])
73 o, g, t = o[ok], g[ok], t[ok]
74 s = t - 7 * np.ceil(g / 7).astype(int) # mismo día de la semana, último ya observado en el origen
75 X = pd.DataFrame({"dia_semana": h.dia_semana[t], "es_quincena": h.es_quincena[t].astype(int),
76 "es_festivo": h.es_festivo[t].astype(int), "semana_anterior": h.y[s] / h.nivel[o],
77 "dias_adelante": g})
78 return X, h.y[t] / h.nivel[o], o, t
